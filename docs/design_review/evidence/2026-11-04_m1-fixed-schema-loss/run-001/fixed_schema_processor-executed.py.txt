"""Explicit evidence-only processor adapter; native schemas, targets and losses remain native."""
from __future__ import annotations

from dataclasses import asdict, replace
from hashlib import sha256
import json

from gliner2.inference.schema import Schema
from gliner2.processing.boundary_preprocessing import build_boundary_batch_metadata
from gliner2.processing.targets import char_span_to_word_boundaries, word_boundaries_to_char_span


def digest(value):
    return sha256(json.dumps(value, ensure_ascii=False, separators=(',', ':'), allow_nan=False).encode()).hexdigest()


def conditioning(batch):
    return dict(input_ids=batch.input_ids.tolist(), attention_mask=batch.attention_mask.tolist(),
        mapped_indices=batch.mapped_indices, text_tokens=batch.text_tokens,
        schema_tokens_list=batch.schema_tokens_list, task_types=batch.task_types,
        schema_counts=batch.schema_counts, original_lengths=batch.original_lengths,
        text_word_indices=batch.text_word_indices.tolist(), text_word_mask=batch.text_word_mask.tolist(),
        text_word_counts=batch.text_word_counts, query_marker_indices=batch.query_marker_indices.tolist(),
        query_marker_mask=batch.query_marker_mask.tolist(), query_group_index=batch.query_group_index.tolist(),
        start_mappings=batch.start_mappings, end_mappings=batch.end_mappings,
        original_texts=batch.original_texts)


class FixedSchemaProcessor:
    """Use the trainer's supported processor seam, without patching native instances."""

    FORMAT = 'm1-fixed-schema-native-targets-experiment-1'

    def __init__(self, native_processor, *, max_gold_per_query=64):
        self.native = native_processor
        self.max_gold_per_query = max_gold_per_query
        self.requested_is_training = False
        self.calls = []
        self.last_receipt = None

    def __getattr__(self, name):
        return getattr(self.native, name)

    def change_mode(self, is_training):
        self.requested_is_training = bool(is_training)
        # Fixed inference preparation and model.train() are independent choices.
        self.native.change_mode(is_training=False)

    def collate_fn_train(self, batch, max_len=None, **kwargs):
        return self._collate(batch, max_len=max_len, build_targets=True,
                             entrypoint='collate_fn_train', **kwargs)

    def collate_fn_inference(self, batch, max_len=None, *, build_targets=False, **kwargs):
        return self._collate(batch, max_len=max_len, build_targets=bool(build_targets),
                             entrypoint='collate_fn_inference', **kwargs)

    def _collate(self, rows, *, max_len, build_targets, entrypoint,
                 architecture='boundary', max_gold_per_query=64,
                 on_capacity_exceeded='raise', error_policy='raise',
                 ignore_missing_entities=False, dedupe_contained_entity_spans=True):
        if max_len is not None or architecture != 'boundary':
            raise ValueError('experiment requires boundary architecture and no truncation')
        if error_policy != 'raise' or on_capacity_exceeded != 'raise' or ignore_missing_entities:
            raise ValueError('experiment requires strict native errors without dropped samples')
        if max_gold_per_query != self.max_gold_per_query:
            raise ValueError('experiment must retain exact checkpoint gold capacity')
        assert rows
        payloads = [row[1] for row in rows]
        for payload in payloads:
            assert payload['format'] == self.FORMAT
            assert payload['family'] in payload['released_schema']['structures']
            assert digest(payload['released_schema']) == payload['released_schema_digest']
            assert digest(payload['target_records']) == payload['target_records_digest']
        prepared = self.native.collate_fn_inference(
            [(text, Schema.from_dict(payload['released_schema'])) for text, payload in rows],
            architecture='boundary', build_targets=False, max_len=None,
            max_gold_per_query=max_gold_per_query, on_capacity_exceeded='raise', error_policy='raise')
        base = conditioning(prepared)
        sample_labels = []
        checks = []
        for index, ((text, payload), specs) in enumerate(zip(rows, prepared.record_specs, strict=True)):
            assert len(specs) == 1 and prepared.task_types[index] == ['json_structures']
            spec = specs[0]
            assert spec.task_name == payload['family']
            declared_fields = payload['released_schema']['structures'][payload['family']]['fields']
            assert [field.name for field in spec.fields] == [field['name'] for field in declared_fields]
            prefix_length = len(prepared.text_tokens[index]) - len(prepared.start_mappings[index])
            prefix = prepared.text_tokens[index][:prefix_length]
            instances, expected = [], []
            for target in payload['target_records']:
                assert list(target) == [field.name for field in spec.fields]
                instance, record_checks = [], []
                for field, declaration in zip(spec.fields, declared_fields, strict=True):
                    assert field.cardinality.value == declaration['cardinality']
                    assert field.exclusive == declaration.get('exclusive', False)
                    value = target[field.name]
                    if value is None:
                        assert field.allows_absent, field.name
                        inclusive, spans = [], []
                    elif 'choices' in value:
                        assert value['choices'] == declaration['choices']
                        assert value['value'] in value['choices']
                        inclusive = [(i, i) for i, token in enumerate(prefix)
                                     if token.lower() == value['value'].lower()]
                        assert inclusive, field.name
                        spans = [(start, end+1) for start, end in inclusive]
                    else:
                        assert text[value['start']:value['end']] == value['text']
                        start, end = char_span_to_word_boundaries(value['start'], value['end'],
                            prepared.start_mappings[index], prepared.end_mappings[index])
                        assert word_boundaries_to_char_span(start, end, prepared.start_mappings[index],
                            prepared.end_mappings[index]) == (value['start'], value['end'])
                        inclusive = [(start+prefix_length, end+prefix_length-1)]
                        spans = [(start+prefix_length, end+prefix_length)]
                    instance.append(inclusive)
                    record_checks.append(dict(role=field.name, spans=spans,
                        absent=value is None, cardinality=field.cardinality.value,
                        exclusive=field.exclusive))
                instances.append(instance)
                expected.append(record_checks)
            sample_labels.append([[len(instances), instances]])
            checks.append(dict(sample_id=payload['sample_id'], family=payload['family'],
                source=payload['source'], expected_records=len(instances),
                expected=expected, prefix=prefix, prefix_length=prefix_length,
                field_order=[field.name for field in spec.fields]))
        if build_targets:
            layouts, targets, record_specs = build_boundary_batch_metadata(
                schema_tokens_list=prepared.schema_tokens_list, task_types=prepared.task_types,
                structure_labels=sample_labels, text_lengths=prepared.text_word_counts,
                record_metadata_list=[schema['record_metadata'] for schema in prepared.original_schemas],
                is_training=False, build_targets=True, max_gold_per_query=max_gold_per_query,
                on_capacity_exceeded='raise', ignore_missing_entities=False,
                dedupe_contained_entity_spans=dedupe_contained_entity_spans,
                original_texts=prepared.original_texts, original_schemas=prepared.original_schemas)
            assert layouts == prepared.query_layouts and record_specs == prepared.record_specs
            result = replace(prepared, query_layouts=layouts, targets=targets,
                             record_specs=record_specs, structure_labels=sample_labels)
            for index, (check, specs) in enumerate(zip(checks, record_specs, strict=True)):
                records = targets.records[index] if targets.records is not None else []
                assert len(records) == check['expected_records']
                assert int(targets.record_targets[2][index].sum()) == len(records)
                assert int(targets.record_targets[4][index].sum()) == len(specs[0].fields)
                for record, expectation in zip(records, check['expected'], strict=True):
                    for field, want in zip(specs[0].fields, expectation, strict=True):
                        actual = [span for alternatives in record.field_for_query(field.query_id).values
                                  for span in alternatives]
                        assert actual == want['spans'], (field.name, actual, want['spans'])
                if not records:
                    assert not bool(targets.mention_mask[index].any())
                check['native_records'] = [asdict(record) for record in records]
        else:
            result = prepared
        assert conditioning(result) == base
        self.last_receipt = dict(entrypoint=entrypoint, requested_is_training=self.requested_is_training,
            native_processor_is_training=self.native.is_training,
            build_targets=build_targets, exact_conditioning=True,
            conditioning=base, conditioning_digest=digest(base),
            samples=checks, strict_capacity=max_gold_per_query,
            on_capacity_exceeded=on_capacity_exceeded,
            original_schemas=result.original_schemas,
            record_specs=[{key: asdict(value) for key,value in specs.items()} for specs in result.record_specs],
            record_mask=result.targets.record_targets[2].tolist() if result.targets is not None else None,
            field_mask=result.targets.record_targets[4].tolist() if result.targets is not None else None,
            mention_mask_count=int(result.targets.mention_mask.sum()) if result.targets is not None else None)
        self.calls.append(dict(entrypoint=entrypoint, samples=[check['sample_id'] for check in checks],
                               conditioning_digest=digest(base)))
        return result
